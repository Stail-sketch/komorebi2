```javascript
function night6TwitterGlitch() {
  if (localStorage.getItem('night6_started')) {
    // 全投稿を「予定通りです」に書き換え
    document.querySelectorAll('.tweet-content').forEach(el => {
      el.textContent = '予定通りです';
    });
    // 吉さんのアカウントを非表示
    document.getElementById('yoshida-account').style.display = 'none';
  }
}
```

## 実装方法

```javascript
// localStorageのフラグを見て投稿を追加
function renderTweets() {
  const tweets = [];
  
  // 初期投稿
  tweets.push(...initialTweets);
  
  // Night1クリア後の投稿
  if (localStorage.getItem('night1_clear')) {
    tweets.push(...night1Tweets);
  }
  // Night2以降も同様
  
  // Night6中の異常
  if (localStorage.getItem('night6_started')) {
    night6TwitterGlitch();
    return;
  }
  
  // タイムライン表示（新しい投稿が上）
  tweets.sort((a, b) => b.timestamp - a.timestamp);
  renderTimeline(tweets);
}
```

## うっきちのタイムライン乗っ取り

特定のページを踏んだ時にlocalStorageにフラグが立ち、Twitterを開くと乗っ取られている。

```javascript
// 乗っ取り状態の確認
if (localStorage.getItem('ukki_takeover')) {
  // 全投稿をうっきちの投稿に置き換え
  // リロードで元に戻るようにフラグを削除
  localStorage.removeItem('ukki_takeover');
}
```
